
# 1. BFS (Breadth First Search)

from collections import deque

def bfs(g, start, goal):
    q = deque([(start, [start])])
    visited = set()

    while q:
        node, path = q.popleft()

        if node == goal:
            return path

        if node not in visited:
            visited.add(node)
            for n in g[node]:
                q.append((n, path + [n]))

g = {'A':['B','C'], 'B':['D'], 'C':['E'], 'D':['G'], 'E':['G'], 'G':[]}
print(bfs(g, 'A', 'G'))


# 2. DFS (Depth First Search)

def dfs(g, start, goal):
    stack = [(start, [start])]
    visited = set()

    while stack:
        node, path = stack.pop()

        if node == goal:
            return path

        if node not in visited:
            visited.add(node)
            for n in g[node]:
                stack.append((n, path + [n]))

g = {'A':['B','C'], 'B':['D'], 'C':['E'], 'D':['G'], 'E':['G'], 'G':[]}
print(dfs(g, 'A', 'G'))


# 3. UCS (Uniform Cost Search)

import heapq

def ucs(g, start, goal):
    q = [(0, start, [start])]
    visited = set()

    while q:
        cost, node, path = heapq.heappop(q)

        if node == goal:
            return path, cost

        if node not in visited:
            visited.add(node)
            for n, c in g[node]:
                heapq.heappush(q, (cost+c, n, path+[n]))

g = {'A':[('B',2),('C',1)], 'B':[('D',2)], 'C':[('E',10)], 'D':[('G',2)], 'E':[('G',1)], 'G':[]}
print(ucs(g, 'A', 'G'))

    visited = set()

# Greedy Best First Search

import heapq

def greedy(g, h, start, goal):
    q = [(h[start], start, [start])]
    visited = []

    while q:
        _, node, path = heapq.heappop(q)

        if node in visited:
            continue

        visited.append(node)

        if node == goal:
            print("Explored Nodes:", visited)
            print("Path:", " -> ".join(path))
            return

        for n in g[node]:
            heapq.heappush(q, (h[n], n, path + [n]))

# Uniform Cost Search

import heapq

def ucs(g, start, goal):
    q = [(0, start, [start])]
    visited = []

    while q:
        cost, node, path = heapq.heappop(q)

        if node in visited:
            continue

        visited.append(node)

        if node == goal:
            print("Explored Nodes:", visited)
            print("Optimal Path:", " -> ".join(path))
            print("Total Travel Cost:", cost)
            return

        for n, c in g[node]:
            heapq.heappush(q, (cost+c, n, path+[n]))
